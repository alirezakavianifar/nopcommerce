<script>
    function initAiSearch() {
        var searchInput = document.getElementById("small-searchterms");
        if (!searchInput) return;
        if (document.getElementById("aiVoiceSearchBtn")) return; // prevent double injection

        // Create AI wrapper and place searchInput inside it
        var wrapper = document.createElement("div");
        wrapper.className = "ai-search-wrapper";
        searchInput.parentNode.insertBefore(wrapper, searchInput);
        wrapper.appendChild(searchInput);

        // Create AI buttons container
        var aiContainer = document.createElement("div");
        aiContainer.className = "ai-search-container";
        aiContainer.innerHTML = `
            <button type="button" class="btn-ai-action" id="aiVoiceSearchBtn" title="Voice Search"><i class="fas fa-microphone"></i></button>
            <button type="button" class="btn-ai-action" id="aiVisualSearchBtn" title="Visual Search"><i class="fas fa-camera"></i></button>
        `;
        wrapper.appendChild(aiContainer);

        // Create Visual Search Results Dropdown
        var resultsDropdown = document.createElement("div");
        resultsDropdown.className = "ai-visual-dropdown";
        resultsDropdown.id = "aiVisualDropdown";
        if (document.documentElement.getAttribute("data-theme") === "dark") {
            resultsDropdown.classList.add("dark-mode");
        }
        resultsDropdown.innerHTML = `
            <div class="ai-visual-header">
                <span>AI Search Results</span>
                <span class="ai-visual-close" id="aiVisualClose">&times;</span>
            </div>
            <div class="ai-visual-items" id="aiVisualItems"></div>
        `;
        wrapper.appendChild(resultsDropdown);

        // Voice Search logic (Web Speech API + Server STT Upload)
        var mediaRecorder;
        var audioChunks = [];
        var voiceBtn = document.getElementById("aiVoiceSearchBtn");
        var voiceOverlay = document.getElementById("aiVoiceOverlay");
        var voiceStopBtn = document.getElementById("aiVoiceStopBtn");
        var voiceStatus = document.getElementById("aiVoiceStatus");

        var SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;

        voiceBtn.addEventListener("click", function() {
            if (SpeechRecognition) {
                // Try Web Speech API (Client-side native local voice search)
                try {
                    var recognition = new SpeechRecognition();
                    recognition.lang = document.documentElement.lang || "fa-IR";
                    recognition.interimResults = false;
                    recognition.maxAlternatives = 1;

                    voiceOverlay.style.display = "flex";
                    voiceStatus.innerText = "Listening...";

                    recognition.onresult = function(event) {
                        voiceOverlay.style.display = "none";
                        var speechResult = event.results[0][0].transcript;
                        if (speechResult) {
                            searchInput.value = speechResult;
                            searchInput.closest("form").submit();
                        }
                    };

                    recognition.onerror = function(event) {
                        console.warn("Web Speech API error, falling back to MediaRecorder upload:", event.error);
                        recognition.stop();
                        startMediaRecorderUpload();
                    };

                    recognition.onend = function() {
                        voiceOverlay.style.display = "none";
                    };

                    voiceStopBtn.onclick = function() {
                        recognition.stop();
                        voiceOverlay.style.display = "none";
                    };

                    recognition.start();
                    return;
                } catch (e) {
                    console.warn("SpeechRecognition init failed, falling back to MediaRecorder upload.");
                }
            }

            startMediaRecorderUpload();
        });

        async function startMediaRecorderUpload() {
            if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
                alert("Microphone access is blocked or unsupported.");
                return;
            }

            try {
                var stream = await navigator.mediaDevices.getUserMedia({ audio: true });
                audioChunks = [];
                mediaRecorder = new MediaRecorder(stream);
                
                mediaRecorder.ondataavailable = function(e) {
                    audioChunks.push(e.data);
                };

                mediaRecorder.onstop = async function() {
                    voiceStatus.innerText = "Transcribing...";
                    var audioBlob = new Blob(audioChunks, { type: 'audio/webm' });
                    
                    var formData = new FormData();
                    formData.append("file", audioBlob, "voice-search.webm");

                    try {
                        var response = await fetch("/AiSearch/VoiceSearch", {
                            method: "POST",
                            body: formData
                        });
                        var result = await response.json();
                        if (result.success) {
                            if (result.query) {
                                searchInput.value = result.query;
                                searchInput.closest("form").submit();
                            } else {
                                alert("No speech was identified.");
                            }
                        } else {
                            alert(result.message || "Transcription failed.");
                        }
                    } catch(e) {
                        alert("Network error during voice search.");
                    } finally {
                        voiceOverlay.style.display = "none";
                        stream.getTracks().forEach(track => track.stop());
                    }
                };

                voiceStopBtn.onclick = function() {
                    if (mediaRecorder && mediaRecorder.state === "recording") {
                        mediaRecorder.stop();
                    }
                };

                voiceOverlay.style.display = "flex";
                voiceStatus.innerText = "Listening...";
                mediaRecorder.start();
            } catch(e) {
                alert("Microphone access is blocked or unsupported.");
            }
        }

        // Visual Search logic
        var visualBtn = document.getElementById("aiVisualSearchBtn");
        var fileInput = document.getElementById("aiVisualFileInput");
        var dropdown = document.getElementById("aiVisualDropdown");
        var dropdownClose = document.getElementById("aiVisualClose");
        var dropdownItems = document.getElementById("aiVisualItems");

        visualBtn.addEventListener("click", function() {
            fileInput.click();
        });

        fileInput.addEventListener("change", async function() {
            if (fileInput.files.length === 0) return;
            
            var file = fileInput.files[0];
            var formData = new FormData();
            formData.append("file", file);

            dropdownItems.innerHTML = '<div class="text-center w-100 py-3"><i class="fas fa-spinner fa-spin fa-2x text-primary"></i><br/><small class="text-muted mt-2 d-block">Analyzing Image...</small></div>';
            dropdown.style.display = "block";

            try {
                var response = await fetch("/AiSearch/VisualSearch", {
                    method: "POST",
                    body: formData
                });
                var result = await response.json();
                if (result.success && result.products && result.products.length > 0) {
                    dropdownItems.innerHTML = "";
                    result.products.forEach(function(product) {
                        var itemEl = document.createElement("a");
                        itemEl.className = "ai-visual-item";
                        itemEl.href = product.url;
                        itemEl.innerHTML = `
                            <img src="${product.pictureUrl}" alt="${product.name}"/>
                            <div class="ai-visual-name">${product.name}</div>
                            <div class="ai-visual-price">${product.price}</div>
                        `;
                        dropdownItems.appendChild(itemEl);
                    });
                } else {
                    dropdownItems.innerHTML = '<div class="text-center w-100 py-3 text-muted">No similar products found.</div>';
                }
            } catch(e) {
                dropdownItems.innerHTML = '<div class="text-center w-100 py-3 text-danger">Search error.</div>';
            }
        });

        dropdownClose.addEventListener("click", function() {
            dropdown.style.display = "none";
        });
    }

    if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", initAiSearch);
    } else {
        initAiSearch();
    }
</script>

        async function getCaptchaToken(action, reCaptchaPublicKey, isReCaptchaV3) {
            var recaptchaToken = '';
            if (isReCaptchaV3) {
                grecaptcha.ready(() => {
                    grecaptcha.execute(reCaptchaPublicKey, { action: action }).then((token) => {
                        recaptchaToken = token;
                    });
                });

                while (recaptchaToken == '') {
                    await new Promise(t => setTimeout(t, 100));
                }
            } else {
                recaptchaToken = $('#newsletter-subscribe-block .captcha-box textarea[name="g-recaptcha-response"').val();
            }
            
            return recaptchaToken;
        }

            loader.innerHTML = '<span class="ai-spinner"></span> ' + (isPersian ? 'در حال پاسخ...' : 'Writing...');
            messages.appendChild(loader);
            messages.scrollTop = messages.scrollHeight;

            try {
                var response = await fetch("/AiChat/SendMessage", {
                    method: "POST",
                    headers: { "Content-Type": "application/json" },
                    body: JSON.stringify(chatHistory)
                });
                var result = await response.json();
                messages.removeChild(loader);

                if (result.success) {
                    renderMessage(result.reply, "assistant");
                    chatHistory.push({ role: "assistant", content: result.reply });

                    if (result.triggerHandoff) {
                        setTimeout(function() {
                            document.getElementById("aiHandoffMessage").value = text; // prefill description
                            handoffOverlay.style.display = "flex";
                        }, 500);
                    }
                } else {
                    renderMessage("Error: Could not retrieve response.", "assistant");
                }
            } catch(e) {
                messages.removeChild(loader);
                renderMessage("Network error.", "assistant");
            }
        }

        function renderMessage(text, role) {
            var msgEl = document.createElement("div");
            msgEl.className = "ai-message " + role;
            msgEl.innerText = text;
            messages.appendChild(msgEl);
            messages.scrollTop = messages.scrollHeight;
        }

        sendBtn.addEventListener("click", sendMessage);
        chatInput.addEventListener("keydown", function(e) {
            if (e.key === "Enter") {
                sendMessage();
            }
        });

        // Handoff submissions
        handoffSubmit.addEventListener("click", async function() {
            var name = document.getElementById("aiHandoffName").value.trim();
            var email = document.getElementById("aiHandoffEmail").value.trim();
            var desc = document.getElementById("aiHandoffMessage").value.trim();

            if (!name || !email || !desc) {
                alert("Please fill in all fields.");
                return;
            }

            handoffSubmit.disabled = true;
            handoffSubmit.innerText = "Submitting...";

            try {
                var response = await fetch(`/AiChat/Handoff?name=${encodeURIComponent(name)}&email=${encodeURIComponent(email)}&message=${encodeURIComponent(desc)}`, {
                    method: "POST"
                });
                var result = await response.json();
                if (result.success) {
                    alert("Your request has been submitted. Our support team will contact you via email.");
                    handoffOverlay.style.display = "none";
                    renderMessage("Your request has been submitted. Our support team will contact you via email.", "assistant");
                } else {
                    alert(result.message || "Failed to submit handoff request.");
                }
            } catch(e) {
                alert("Network error.");
            } finally {
                handoffSubmit.disabled = false;
                handoffSubmit.innerText = "Submit Handoff Request";
            }
        });

        handoffCancel.addEventListener("click", function() {
            handoffOverlay.style.display = "none";
        });
    });
