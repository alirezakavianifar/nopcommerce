\section{معرفی ماژول‌های هوشمند و سرویس‌های پیشرفته}
سیستم nopCommerce به مجموعه‌ای از ماژول‌های هوش مصنوعی پیشرفته مجهز گردیده است:
\begin{itemize}[label=\small$\bullet$]
    \item \textbf{ماژول B (جستجوی هوشمند متنی، صوتی و تصویری):}
    شامل جستجوی معنایی متنی مبتنی بر بردار ویژگی‌ها (\lr{Text Embeddings})، جستجوی صوتی به زبان فارسی (\lr{Speech-to-Text}) و جستجوی تصویری کالاها (\lr{Visual Search}) به همراه \lr{REST API} اختصاصی.
    \item \textbf{ماژول C (تشخیص خودکار کالاهای تکراری با هوش مصنوعی):}
    اسکن خودکار تصاویر و ویژگی‌های فنی کاتالوگ فروشندگان و تشکیل صف بررسی هوشمند (\lr{Duplicate Queue}) در پنل مدیریت.
    \item \textbf{ماژول بازاریابی و کاتالوگ فروشندگان (\lr{Seller Marketing}):}
    ارسال و مدیریت درخواست‌های تبلیغاتی و محصولات جدید توسط فروشندگان از طریق پنل و وب‌سرویس \lr{REST}.
    \item \textbf{سیستم خرید گروهی و پاداش لیدری (\lr{Group Purchase \& Rewards}):}
    موتور مستقل محاسبه و واریز پاداش به کیف پول و تخصیص بلیط قرعه‌کشی بدون تداخل با تخفیف‌های سبد خرید.
    \item \textbf{ماژول امنیت، تایید دو مرحله‌ای و محدودیت دستگاه (\lr{SMS 2FA \& Device Binding}):}
    ورود امن پیامکی و احراز هویت سخت‌افزاری بر پایه \lr{Device Token} و لیست سفید \lr{IP}.
    \item \textbf{موتور ارسال مشروط ۳ سطحی (\lr{3-Tier Conditional Shipping}):}
    ارزیابی اولویت‌بندی ارسال بر اساس پوشش شهری، نوع محصول و انبار با ۴ شیوه ارسال.
\end{itemize}

\section{پیش‌نیازهای سیستم و تنظیمات زبان فارسی}
برای کارکرد صحیح افزونه‌ها و نمایش کامل محیط راست‌چین:
\begin{enumerate}
    \item \textbf{تنظیم زبان پیش‌فرض:} در پنل مدیریت به مسیر \lr{Configuration > Languages} مراجعه نموده و زبان فارسی (\lr{fa-IR}) را به عنوان زبان پیش‌فرض فعال کنید.
    \item \textbf{بومی‌سازی و قلم استاندارد:} تمامی بخش‌های اختصاصی به ترجمه جامع فارسی مجهز شده و از قلم استاندارد **وزیرمتن (Vazirmatn)** بهره می‌برند.
\end{enumerate}

\begin{itemize}[label=\small$\bullet$]
    \item \textbf{ارسال درخواست تبلیغاتی یا کاتالوگ جدید:} \lr{POST /api/seller-marketing/submit} \\
    دریافت پارامترهای کالا (\lr{productId} یا اطلاعات کامل کالا)، بودجه روزانه تبلیغات (\lr{dailyBudget})، یادداشت و تصویر.

    بازگرداندن لیست درخواست‌های جاری، تاییدشده یا ردشده به همراه توضیحات بازخورد مدیر ارشد.
\end{itemize}

\section{بخش چهارم: راهنمای ماژول‌های هوش مصنوعی (ماژول B و C)}

\subsection{گام ۱۲: ماژول B (جستجوی هوشمند متنی، صوتی و تصویری)}
ماژول جستجوی هوشمند \lr{ArtificialIntelligence} سه شیوه نوین جستجو را در اختیار کاربران و سامانه‌های خارجی قرار می‌دهد:
\begin{enumerate}
    \item \textbf{جستجوی معنایی متنی هوش مصنوعی (\lr{AI Semantic Text Search}):} \\
    در این روش، عبارت جستجو شده توسط کاربر به یک بردار ۱۵۳۶ بعدی (\lr{Embedding Vector}) تبدیل شده و با بردار ویژگی‌ها، نام و مشخصات فنی تمامی کالاها مقایسه کسینوسی (\lr{Cosine Similarity}) می‌شود. این فناوری امکان درک معنایی و جستجوی مفهومی کالاها را فراهم می‌سازد.
    \item \textbf{جستجوی تصویری هوش مصنوعی (\lr{Visual Search}):} \\
    کاربر تصویری از محصول را آپلود کرده و هوش مصنوعی با استخراج ویژگی‌های بصری، کالاهای مشابه را نمایش می‌دهد.
    \item \textbf{جستجوی صوتی هوش مصنوعی (\lr{Voice Search}):} \\
    صوت ضبط‌شده کاربر با مدل تشخیص گفتار فارسی (\lr{Speech-to-Text}) تحلیل و نتایج کالاها مستقیماً بازگردانده می‌شود.
\end{enumerate}

\subsubsection*{سرویس‌های \lr{REST API} جستجوی هوشمند}
برای اپلیکیشن‌های موبایل، تبلت‌های انبارداری و پیک‌ها، اندپوینت‌های استاندارد \lr{RESTful} زیر ارائه شده است:
\begin{itemize}[label=\small$\bullet$]
    \item \textbf{جستجوی معنایی متنی:} \lr{POST /api/ai/text-search} یا \lr{GET /api/ai/text-search?query=...}
    \item \textbf{جستجوی تصویری:} \lr{POST /api/ai/visual-search} (ارسال فایل تصویر در قالب \lr{multipart/form-data})
    \item \textbf{جستجوی صوتی:} \lr{POST /api/ai/voice-search} (ارسال فایل صوتی ضبط‌شده)
\end{itemize}

\subsection{گام ۱۵: امنیت دستگاه و لیست سفید \lr{IP} (\lr{Device Token Security})}
\noindent \textbf{توضیح فنی معماری امنیت وب:} \\

\section{بخش ششم: راهنمای موتور ارسال مشروط ۳ سطحی}

\subsection{گام ۱۶: مدیریت روش‌های ارسال مشروط (\lr{Conditional Shipping})}
موتور ارسال مشروط، روش‌های مجاز حمل را با ارزیابی ۳ شرط متوالی در خط لوله محاسباتی فیلتر می‌کند:
\begin{equation*}
    \text{۱. پوشش جغرافیایی شهر} \longrightarrow \text{۲. پشتیبانی نوع محصول} \longrightarrow \text{۳. موجودی و تجهیزات انبار}
\end{equation*}

روش‌های ۴‌گانه ارسال شامل موارد زیر می‌باشند:
\begin{enumerate}
    \item \textbf{پیک اختصاصی درون‌شهری (\lr{Courier}):} تحویل سریع در محدوده شهر فعال.
    \item \textbf{باربری بین‌شهری (\lr{Freight}):} مخصوص بارهای حجیم یا سفارش‌های عمده.
    \item \textbf{کارگو و حمل هوایی (\lr{Cargo}):} ارسال فوری بین‌استانی به شهرهای دارای فرودگاه.
    \item \textbf{پست پیشتاز/اکسپرس (\lr{Express}):} پوشش سراسری کلیه شهرها و روستاها.
\end{enumerate}

\end{figure}
